## 六、服务注册与发现


### 6.1 etcd 注册

```go
import (
    "go-micro.dev/v5"
    "go-micro.dev/v5/registry"
    "go-micro.dev/v5/registry/etcd"
)

func main() {
    // 初始化 etcd registry
    etcdRegistry := etcd.NewRegistry(
        registry.Addrs("etcd-host:2379"),
        etcd.TLSConfig(tlsConfig),  // 生产环境必须启用 TLS
    )

    // 创建服务
    service := micro.NewService(
        micro.Name("user-service"),
        micro.Address(":8080"),
        micro.Registry(etcdRegistry),
    )

    service.Init()
    // 注册 Handler ...
    service.Run()
}
```

### 6.2 HTTP Gateway

```go
func startHTTPGateway(ctx context.Context, grpcAddr, httpAddr string) error {
    mux := runtime.NewServeMux(
        runtime.WithMarshalerOption(runtime.MIMEWildcard, &CustomMarshaler{}),
        runtime.WithErrorHandler(customErrorHandler),
    )

    opts := []grpc.DialOption{grpc.WithTransportCredentials(insecure.NewCredentials())}

    // 注册各服务的 HTTP 路由
    if err := pb.RegisterUserServiceGw(ctx, mux, grpcAddr, opts...); err != nil {
        return err
    }

    return http.ListenAndServe(httpAddr, mux)
}
```
